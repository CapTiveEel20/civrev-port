# Sid Meier's Civilization Revolution (PS3, BLES00238) -- native PC port on the
# shared ps3recomp runtime.
#
# Layout follows sp00nznet/tornado (MIT): the lifted game code plus ps3recomp's
# generic PPU boot harness, linked against a prebuilt ps3recomp_runtime.lib.
#
# Generate the sources first (tools\relift.ps1), then build (build.ps1).
# Toolchain: clang-cl. The harness and the lifted code use __builtin_bswap,
# __int128 and weak symbols, none of which plain MSVC cl provides.
cmake_minimum_required(VERSION 3.20)

# Default to Release. CMake+Ninja pick no optimisation when no type is given,
# and lifted code built without optimisation runs about 3x slower (Tornado notes).
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "build type" FORCE)
endif()

project(civrev CXX C)

# clang-cl with no /O flag means no optimisation, so -O2 is explicit.
# Lifted code type-puns guest memory (-fno-strict-aliasing) and must not fuse
# multiply-adds the PS3 CPU did not (-ffp-contract=off). clang-cl silently
# ignores a bare -ffp-contract=off ("unknown argument ignored"), so it has to
# go through /clang: to reach the compiler. Tornado's CMakeLists passes it bare.
add_compile_options($<$<CONFIG:Release,RelWithDebInfo>:-O2> -fno-strict-aliasing /clang:-ffp-contract=off)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 17)

# Default: a ps3recomp checkout next to this project (../ps3recomp).
# Override with -DPS3RECOMP_DIR=<path> or the PS3RECOMP_DIR environment variable.
if(DEFINED ENV{PS3RECOMP_DIR})
    set(_ps3recomp_default "$ENV{PS3RECOMP_DIR}")
else()
    get_filename_component(_ps3recomp_default "${CMAKE_CURRENT_SOURCE_DIR}/../ps3recomp" ABSOLUTE)
endif()
set(PS3RECOMP_DIR "${_ps3recomp_default}" CACHE PATH "ps3recomp repo root")
if(NOT EXISTS "${PS3RECOMP_DIR}/runtime/ppu/tests/boot_main.cpp")
    message(FATAL_ERROR "ps3recomp not found at '${PS3RECOMP_DIR}'. Pass -DPS3RECOMP_DIR=<path>.")
endif()
set(PS3RECOMP_RUNTIME_LIB "${PS3RECOMP_DIR}/build-release/ps3recomp_runtime.lib"
    CACHE FILEPATH "prebuilt ps3recomp runtime library")

# Generated by tools\relift.ps1 and git-ignored: lifted PPU code, HLE NID table,
# lifted SPU programs.
file(GLOB RECOMP_SOURCES "src/recomp/*.cpp" "src/recomp/*.c")
file(GLOB GEN_SOURCES    "src/gen/*.cpp" "src/gen/*.c")
file(GLOB_RECURSE SPU_SOURCES "src/spu_gen/*.c")
if(NOT RECOMP_SOURCES)
    message(FATAL_ERROR "src/recomp/ is empty -- run tools\\relift.ps1 first")
endif()

add_executable(civrev
    "${PS3RECOMP_DIR}/runtime/ppu/tests/boot_main.cpp"   # generic boot harness
    "${PS3RECOMP_DIR}/runtime/ppu/ppu_loader.cpp"        # memory + ELF loader + run
    "${PS3RECOMP_DIR}/runtime/ppu/ppu_hle.cpp"           # NID -> handler dispatch
    "${PS3RECOMP_DIR}/runtime/ppu/ppu_sysprx.cpp"        # boot-critical CRT
    "${PS3RECOMP_DIR}/runtime/ppu/ppu_fs.cpp"            # cellFs virtual file system
    ${GEN_SOURCES}
    ${SPU_SOURCES}
    ${RECOMP_SOURCES}
)

file(GLOB LIB_INC_DIRS LIST_DIRECTORIES true "${PS3RECOMP_DIR}/libs/*")
target_include_directories(civrev PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/src/compat"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp"
    "${PS3RECOMP_DIR}/runtime/ppu"
    "${PS3RECOMP_DIR}/runtime/spu"
    "${PS3RECOMP_DIR}/runtime/prx"
    "${PS3RECOMP_DIR}/include"
    ${LIB_INC_DIRS}
)

target_link_libraries(civrev PRIVATE "${PS3RECOMP_RUNTIME_LIB}")
if(WIN32)
    target_link_libraries(civrev PRIVATE ws2_32 bcrypt xinput ole32 d3d12 dxgi dxguid d3dcompiler)
    # Every guest call is a host call, so recompiled call chains nest deeply: 256 MB stack.
    target_link_options(civrev PRIVATE /STACK:268435456 /MAP:civrev.map)
    # Lifted translation units are huge; /bigobj lifts the object-file section limit.
    target_compile_options(civrev PRIVATE /bigobj)
endif()
